cmake_minimum_required(VERSION 3.25)
project(post_profit_exchange VERSION 0.3.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CTest)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS configs/exchange.cfg)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/configs/exchange.cfg" PH_EXCHANGE_DEFAULTS)
string(FIND "${PH_EXCHANGE_DEFAULTS}" ")PHXCFG\"" PH_EXCHANGE_DELIMITER)
if(NOT PH_EXCHANGE_DELIMITER EQUAL -1)
  message(FATAL_ERROR "Configuration contains reserved C++ embedding delimiter")
endif()
configure_file(simulation/default_config.hpp.in generated/default_config.hpp @ONLY)
if(NOT TARGET ph_price_enumeration)
  add_subdirectory(../../tools/price-optimization price-optimization)
endif()
if(NOT TARGET ph::exponential_smoothing)
  add_subdirectory(../../tools/exponential-smoothing exponential-smoothing)
endif()
add_library(ph_exchange_models simulation/models.cpp)
add_library(ph::exchange_models ALIAS ph_exchange_models)
target_include_directories(ph_exchange_models PUBLIC simulation)
target_include_directories(ph_exchange_models SYSTEM PUBLIC ../../tools/price-optimization/third_party)
target_link_libraries(ph_exchange_models PUBLIC ph::exponential_smoothing)
add_library(ph_exchange_simulation simulation/simulation.cpp)
target_include_directories(ph_exchange_simulation PUBLIC simulation)
target_include_directories(ph_exchange_simulation PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_include_directories(ph_exchange_simulation SYSTEM PUBLIC ../../tools/price-optimization/third_party)
target_link_libraries(ph_exchange_simulation PUBLIC ph::price_enumeration ph::exchange_models)
if(MSVC)
  target_compile_options(ph_exchange_simulation PRIVATE /W4)
else()
  target_compile_options(ph_exchange_simulation PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(exchange_simulation simulation/main.cpp simulation/files.cpp)
target_include_directories(exchange_simulation PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_link_libraries(exchange_simulation PRIVATE ph_exchange_simulation)
if(BUILD_TESTING)
  add_executable(exchange_files_test simulation/files_test.cpp simulation/files.cpp)
  target_include_directories(exchange_files_test PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
  target_compile_definitions(exchange_files_test PRIVATE PH_EXCHANGE_SOURCE_CONFIG="${CMAKE_CURRENT_SOURCE_DIR}/configs/exchange.cfg")
  target_link_libraries(exchange_files_test PRIVATE ph_exchange_simulation)
  add_test(NAME exchange_files COMMAND exchange_files_test)
  add_executable(exchange_models_test simulation/models_test.cpp)
  target_link_libraries(exchange_models_test PRIVATE ph::exchange_models)
  add_test(NAME exchange_models COMMAND exchange_models_test)
  add_executable(exchange_simulation_test simulation/simulation_test.cpp)
  target_include_directories(exchange_simulation_test SYSTEM PRIVATE ../../tools/price-optimization/third_party)
  target_link_libraries(exchange_simulation_test PRIVATE ph_exchange_simulation)
  add_test(NAME exchange_simulation COMMAND exchange_simulation_test)
  set_tests_properties(exchange_simulation PROPERTIES TIMEOUT 30)
endif()
