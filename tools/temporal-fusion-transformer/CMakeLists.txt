cmake_minimum_required(VERSION 3.25)
project(praesidium-humanitatis VERSION 0.1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

find_package(Torch 2.6 REQUIRED)
option(PRAESIDIUM_HUMANITATIS_CUDA_TESTS "Register tests that require an available CUDA device" OFF)

add_library(praesidium_humanitatis_tft src/tft.cpp src/checkpoint.cpp src/evaluation.cpp src/calibration.cpp)
add_library(praesidium_humanitatis::tft ALIAS praesidium_humanitatis_tft)
target_include_directories(praesidium_humanitatis_tft PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_link_libraries(praesidium_humanitatis_tft PUBLIC "${TORCH_LIBRARIES}")
separate_arguments(TORCH_CXX_FLAGS_LIST NATIVE_COMMAND "${TORCH_CXX_FLAGS}")
target_compile_options(praesidium_humanitatis_tft PUBLIC ${TORCH_CXX_FLAGS_LIST})
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(praesidium_humanitatis_tft PRIVATE -Wall -Wextra -Wpedantic)
endif()

add_executable(tft_demo examples/train_synthetic.cpp)
target_link_libraries(tft_demo PRIVATE praesidium_humanitatis::tft)
add_executable(tft_uncertainty examples/validate_uncertainty.cpp)
target_link_libraries(tft_uncertainty PRIVATE praesidium_humanitatis::tft)
add_executable(tft_compare_reference examples/compare_reference.cpp)
target_link_libraries(tft_compare_reference PRIVATE praesidium_humanitatis::tft)
add_executable(tft_train_electricity examples/train_electricity.cpp)
target_link_libraries(tft_train_electricity PRIVATE praesidium_humanitatis::tft)

include(CTest)
if(BUILD_TESTING)
  add_executable(tft_test tests/tft_test.cpp)
  target_link_libraries(tft_test PRIVATE praesidium_humanitatis::tft)
  foreach(suite numerics reference learning long_range)
    add_executable(tft_${suite}_test tests/tft_${suite}_test.cpp)
    target_link_libraries(tft_${suite}_test PRIVATE praesidium_humanitatis::tft)
    add_test(NAME tft_${suite}_cpu COMMAND tft_${suite}_test --device cpu)
    set_tests_properties(tft_${suite}_cpu PROPERTIES TIMEOUT 180)
    if(PRAESIDIUM_HUMANITATIS_CUDA_TESTS)
      add_test(NAME tft_${suite}_cuda COMMAND tft_${suite}_test --device cuda)
      set_tests_properties(tft_${suite}_cuda PROPERTIES TIMEOUT 300 SKIP_RETURN_CODE 77)
    endif()
  endforeach()
  set_tests_properties(tft_long_range_cpu PROPERTIES TIMEOUT 300)
  if(PRAESIDIUM_HUMANITATIS_CUDA_TESTS)
    set_tests_properties(tft_long_range_cuda PROPERTIES TIMEOUT 600)
  endif()
  add_test(NAME tft_cpu COMMAND tft_test --device cpu)
  set_tests_properties(tft_cpu PROPERTIES TIMEOUT 180)
  add_executable(checkpoint_test tests/checkpoint_test.cpp)
  target_link_libraries(checkpoint_test PRIVATE praesidium_humanitatis::tft)
  add_executable(evaluation_test tests/evaluation_test.cpp)
  target_link_libraries(evaluation_test PRIVATE praesidium_humanitatis::tft)
  add_executable(calibration_test tests/calibration_test.cpp)
  target_link_libraries(calibration_test PRIVATE praesidium_humanitatis::tft)
  add_executable(compatibility_test tests/compatibility_test.cpp)
  target_link_libraries(compatibility_test PRIVATE praesidium_humanitatis::tft)
  add_test(NAME compatibility_cpu COMMAND compatibility_test --device cpu)
  set_tests_properties(compatibility_cpu PROPERTIES TIMEOUT 180)
  add_test(NAME checkpoint_cpu COMMAND checkpoint_test --device cpu)
  add_test(NAME evaluation_cpu COMMAND evaluation_test --device cpu)
  add_test(NAME calibration_cpu COMMAND calibration_test --device cpu)
  set_tests_properties(calibration_cpu PROPERTIES TIMEOUT 180)
  set_tests_properties(checkpoint_cpu evaluation_cpu PROPERTIES TIMEOUT 180)
  add_test(NAME demo_workflow_cpu COMMAND ${CMAKE_COMMAND}
    -DDEMO=$<TARGET_FILE:tft_demo> -DWORK_DIR=${CMAKE_CURRENT_BINARY_DIR}/demo-workflow
    -P ${CMAKE_CURRENT_SOURCE_DIR}/tests/demo_workflow.cmake)
  set_tests_properties(demo_workflow_cpu PROPERTIES TIMEOUT 180)
  if(PRAESIDIUM_HUMANITATIS_CUDA_TESTS)
    add_test(NAME compatibility_cuda COMMAND compatibility_test --device cuda)
    set_tests_properties(compatibility_cuda PROPERTIES TIMEOUT 180 SKIP_RETURN_CODE 77)
    add_test(NAME tft_cuda COMMAND tft_test --device cuda)
    set_tests_properties(tft_cuda PROPERTIES TIMEOUT 180 SKIP_RETURN_CODE 77)
    add_test(NAME checkpoint_cuda COMMAND checkpoint_test --device cuda)
    add_test(NAME evaluation_cuda COMMAND evaluation_test --device cuda)
    add_test(NAME calibration_cuda COMMAND calibration_test --device cuda)
    set_tests_properties(calibration_cuda PROPERTIES TIMEOUT 180 SKIP_RETURN_CODE 77)
    set_tests_properties(checkpoint_cuda evaluation_cuda PROPERTIES TIMEOUT 180 SKIP_RETURN_CODE 77)
  endif()
endif()
