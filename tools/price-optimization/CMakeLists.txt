cmake_minimum_required(VERSION 3.25)
project(praesidium_price_optimization VERSION 0.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

option(PH_PRICE_WITH_AMPL "Build the AMPL C++ API backend (vendor dependencies required)" OFF)
set(AMPLAPI_ROOT "" CACHE PATH "Staged AMPL C++ API SDK root (include and lib)")
set(PH_AMPL_BINARY_DIR "" CACHE PATH "AMPL runtime directory; nonempty enables licensed integration CTest")
set(PH_AMPL_SOLVER "highs" CACHE STRING "AMPL integration solver (only highs is currently supported)")
set_property(CACHE PH_AMPL_SOLVER PROPERTY STRINGS highs)
if(PH_PRICE_WITH_AMPL AND NOT PH_AMPL_SOLVER STREQUAL "highs")
  message(FATAL_ERROR "Only highs has a reviewed status-code policy in this initial AMPL adapter")
endif()

add_library(ph_price_core src/engine.cpp src/ampl_data.cpp)
target_include_directories(ph_price_core PUBLIC include)
add_library(ph::price_core ALIAS ph_price_core)

add_library(ph_price_enumeration src/enumeration.cpp)
target_link_libraries(ph_price_enumeration PUBLIC ph_price_core)
add_library(ph::price_enumeration ALIAS ph_price_enumeration)

add_library(ph_price_integration src/integration.cpp)
target_link_libraries(ph_price_integration PUBLIC ph_price_core)
target_include_directories(ph_price_integration SYSTEM PUBLIC third_party)
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/models/public_prices.mod" PH_PRICE_MODEL_SHA256)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS models/public_prices.mod)
target_compile_definitions(ph_price_integration PRIVATE PH_PRICE_MODEL_SHA256="${PH_PRICE_MODEL_SHA256}")
add_library(ph::price_integration ALIAS ph_price_integration)

add_library(ph_price_ampl src/ampl_backend.cpp)
target_link_libraries(ph_price_ampl PUBLIC ph_price_core)
add_library(ph::price_ampl ALIAS ph_price_ampl)
if(PH_PRICE_WITH_AMPL)
  find_path(PH_AMPL_INCLUDE_DIR NAMES ampl/ampl.h HINTS "${AMPLAPI_ROOT}/include" REQUIRED)
  find_library(PH_AMPL_LIBRARY NAMES ampl ampl64 HINTS "${AMPLAPI_ROOT}"
    PATH_SUFFIXES lib lib/amd64 lib/x86_64 lib/x86_64-linux-gnu bin REQUIRED)
  file(READ "${CMAKE_CURRENT_SOURCE_DIR}/models/public_prices.mod" PH_PRICE_MODEL_TEXT)
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS models/public_prices.mod)
  configure_file(src/model_text.hpp.in generated/model_text.hpp @ONLY)
  target_include_directories(ph_price_ampl SYSTEM PRIVATE "${PH_AMPL_INCLUDE_DIR}")
  target_include_directories(ph_price_ampl PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
  target_compile_definitions(ph_price_ampl PRIVATE PH_PRICE_WITH_AMPL=1)
  target_link_libraries(ph_price_ampl PRIVATE "${PH_AMPL_LIBRARY}")
  add_executable(price_ampl_demo examples/ampl_demo.cpp)
  target_link_libraries(price_ampl_demo PRIVATE ph_price_ampl)
elseif(PH_AMPL_BINARY_DIR)
  message(FATAL_ERROR "PH_AMPL_BINARY_DIR requires PH_PRICE_WITH_AMPL=ON; no fallback integration test")
endif()

if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  add_library(ph_price_process src/process.cpp)
  target_include_directories(ph_price_process PUBLIC include)
  add_executable(price_ampl_worker src/ampl_worker.cpp)
  target_link_libraries(price_ampl_worker PRIVATE ph_price_ampl ph_price_integration)
  add_executable(price_backend src/backend.cpp)
  target_link_libraries(price_backend PRIVATE ph_price_integration ph_price_process)
endif()

foreach(target ph_price_core ph_price_ampl ph_price_integration ph_price_enumeration)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
  endif()
endforeach()

add_executable(price_guardrail_demo examples/guardrail_demo.cpp)
target_link_libraries(price_guardrail_demo PRIVATE ph_price_core)

include(CTest)
if(BUILD_TESTING)
  add_executable(price_enumeration_test tests/enumeration_test.cpp)
  target_link_libraries(price_enumeration_test PRIVATE ph_price_enumeration)
  target_include_directories(price_enumeration_test PRIVATE examples)
  add_test(NAME price_enumeration COMMAND price_enumeration_test)
  set_tests_properties(price_enumeration PROPERTIES TIMEOUT 20)
  add_executable(price_core_test tests/core_test.cpp)
  target_link_libraries(price_core_test PRIVATE ph_price_core ph_price_ampl)
  target_include_directories(price_core_test PRIVATE src examples)
  add_test(NAME price_core COMMAND price_core_test)
  add_test(NAME price_guardrail_demo COMMAND price_guardrail_demo)
  set_tests_properties(price_core price_guardrail_demo PROPERTIES TIMEOUT 20)
  add_executable(price_contract_test tests/integration_contract_test.cpp)
  target_link_libraries(price_contract_test PRIVATE ph_price_integration)
  add_test(NAME price_contract COMMAND price_contract_test)
  set_tests_properties(price_contract PROPERTIES TIMEOUT 20)
  if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    add_executable(price_process_fixture tests/process_fixture.cpp)
    add_executable(price_process_test tests/process_test.cpp)
    target_link_libraries(price_process_test PRIVATE ph_price_process)
    add_test(NAME price_process COMMAND price_process_test $<TARGET_FILE:price_process_fixture>)
    set_tests_properties(price_process PROPERTIES TIMEOUT 30)
    add_executable(price_backend_fixture tests/backend_fixture.cpp)
    target_link_libraries(price_backend_fixture PRIVATE ph_price_integration)
    add_executable(price_backend_test tests/backend_test.cpp)
    target_link_libraries(price_backend_test PRIVATE ph_price_integration)
    add_test(NAME price_backend_protocol COMMAND price_backend_test
      $<TARGET_FILE:price_backend> $<TARGET_FILE:price_backend_fixture>)
    set_tests_properties(price_backend_protocol PROPERTIES TIMEOUT 30)
    if(PH_PRICE_WITH_AMPL AND PH_AMPL_BINARY_DIR)
      add_test(NAME price_backend_ampl COMMAND price_backend_test
        $<TARGET_FILE:price_backend> $<TARGET_FILE:price_backend_fixture>
        $<TARGET_FILE:price_ampl_worker> "${PH_AMPL_BINARY_DIR}")
      set_tests_properties(price_backend_ampl PROPERTIES TIMEOUT 60 LABELS "ampl;licensed")
    elseif(NOT PH_PRICE_WITH_AMPL)
      add_test(NAME price_backend_unavailable COMMAND price_backend_test
        $<TARGET_FILE:price_backend> $<TARGET_FILE:price_backend_fixture>
        $<TARGET_FILE:price_ampl_worker>)
      set_tests_properties(price_backend_unavailable PROPERTIES TIMEOUT 30)
    endif()
  endif()
  if(PH_PRICE_WITH_AMPL)
    add_executable(price_ampl_test tests/ampl_test.cpp)
    target_link_libraries(price_ampl_test PRIVATE ph_price_ampl ph_price_enumeration)
    target_include_directories(price_ampl_test PRIVATE examples)
    if(PH_AMPL_BINARY_DIR)
      add_test(NAME price_ampl_integration COMMAND price_ampl_test "${PH_AMPL_BINARY_DIR}" "${PH_AMPL_SOLVER}")
      set_tests_properties(price_ampl_integration PROPERTIES TIMEOUT 120 LABELS "ampl;licensed")
    endif()
  endif()
endif()
